OPCODE (OP3 OP2 OP1 OP0)
0000SET RAM[A] = D
0001MOV RAM[A] = RAM[B]
0010LINK RAM[A] = (B & F0)|(C & 0F)
0011ADD RAM[A] = B + C
0100SUB RAM[A] = B - C
0101MUL RAM[A] = B * C
0110DIV RAM[A] = B / C
0111AND RAM[A] = B & C
1000OR RAM[A] = B | C
1001XOR RAM[A] = B ^ C
1010NOT RAM[A] = ~B
1011SHL RAM[A] = B << 1
1100SHR RAM[A] = B >> 1
1101INC RAM[A] = B + 1
1110DEC RAM[A] = B - 1
1111NEG RAM[A] = 0 - B
ADDRESS
B = T3..T0 (low 4 of tag)
C = D3..D0 (low 4 of data)
A = A3..A0 (dest)
AKSI
WR=1 tulis · RD=1 baca (status)
